===============================================================================
📡 ACCESSING YOUR MOCK SERVER
===============================================================================

{{- if eq .Values.service.type "ClusterIP" }}

1. From within the cluster:
   
   Service DNS: {{ .Release.Name }}-prism.{{ .Release.Namespace }}.svc.cluster.local
   Port: {{ .Values.service.port }}

   Example:
   curl http://{{ .Release.Name }}-prism.{{ .Release.Namespace }}.svc.cluster.local:{{ .Values.service.port }}/health

2. Using port-forward (for local testing):

   kubectl port-forward svc/{{ .Release.Name }}-prism 8080:{{ .Values.service.port }} -n {{ .Release.Namespace }}

   Then access: http://localhost:8080

{{- else if eq .Values.service.type "NodePort" }}

NodePort Service:

   export NODE_PORT=$(kubectl get svc {{ .Release.Name }}-prism -n {{ .Release.Namespace }} -o jsonpath='{.spec.ports[0].nodePort}')
   export NODE_IP=$(kubectl get nodes -o jsonpath='{.items[0].status.addresses[?(@.type=="ExternalIP")].address}')
   
   echo "Access your mock server at: http://$NODE_IP:$NODE_PORT"

{{- else if eq .Values.service.type "LoadBalancer" }}

LoadBalancer Service:

   export SERVICE_IP=$(kubectl get svc {{ .Release.Name }}-prism -n {{ .Release.Namespace }} -o jsonpath='{.status.loadBalancer.ingress[0].ip}')
   
   echo "Access your mock server at: http://$SERVICE_IP:{{ .Values.service.port }}"
   
   Note: It may take a few minutes for the LoadBalancer IP to be available.

{{- end }}
